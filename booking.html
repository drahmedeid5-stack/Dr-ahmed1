// ============================================= // الكود الصحيح لـ booking.html (جافاسكريبت) // ============================================= // رابط API بتاعك (نفس الرابط اللي خدته من Apps Script) const API_URL = 'https://script.google.com/macros/s/AKfycbzUHh_JfDAR6P9KxKxS9yNTuFPu7eRDarHV5ojwNwI7dPUj3hOFNsm7PZZwz0ywu1RITQ/exec'; // ===== دالة جلب مواعيد العمل من الشيت ===== async function getSchedule() { try { const response = await fetch(`${API_URL}?action=getSchedule`); const data = await response.json(); return data; } catch (error) { console.error('خطأ في جلب المواعيد:', error); return null; } } // ===== دالة تحديث قائمة الأيام والمواعيد حسب الفرع ===== async function updateAvailableSlots(branchName) { const daySelect = document.getElementById('day'); const timeSelect = document.getElementById('time'); // تفريغ القوائم الحالية daySelect.innerHTML = ''; timeSelect.innerHTML = ''; // جلب جدول المواعيد من الشيت const schedule = await getSchedule(); if (!schedule) { alert('لا يمكن جلب مواعيد العمل. تأكد من الاتصال بالشيت.'); return; } // تحديد مفتاح البحث حسب الفرع let daysKey = ''; let timesKey = ''; if (branchName === 'أكتوبر') { daysKey = 'days_october'; timesKey = 'times_october'; } else if (branchName === 'فيصل') { daysKey = 'days_faisal'; timesKey = 'times_faisal'; } else { return; } const daysValue = schedule[daysKey] || ''; const timesValue = schedule[timesKey] || ''; // لو مفيش أيام محددة أو مكتوب "مغلق" if (!daysValue || daysValue.toLowerCase().includes('مغلق')) { daySelect.innerHTML = ''; return; } // تقسيم الأيام والمواعيد const daysList = daysValue.split(',').map(item => item.trim()); const timesList = timesValue.split(',').map(item => item.trim()); // إضافة الأيام للقائمة المنسدلة daysList.forEach(day => { if (day) { const option = document.createElement('option'); option.value = day; option.textContent = day; daySelect.appendChild(option); } }); // إضافة المواعيد للقائمة المنسدلة function updateTimes() { timeSelect.innerHTML = ''; timesList.forEach(time => { if (time) { const option = document.createElement('option'); option.value = time; option.textContent = time; timeSelect.appendChild(option); } }); } // أول مرة تظهر المواعيد مع أول يوم updateTimes(); // عند تغيير اليوم، نظهر المواعيد تاني (نفسها) daySelect.onchange = function() { updateTimes(); }; } // ===== دالة إرسال الحجز ===== async function submitBooking(e) { e.preventDefault(); const branch = document.getElementById('branch'); const branchName = branch.value; const branchPhone = branch.options[branch.selectedIndex].getAttribute('data-phone'); const day = document.getElementById('day').value; const time = document.getElementById('time').value; const name = document.getElementById('name').value.trim(); const phone = document.getElementById('phone').value.trim(); const notes = document.getElementById('notes').value.trim(); if (!branchName || !day || !time || !name || !phone) { alert('يرجى ملء جميع الحقول المطلوبة.'); return; } // إرسال البيانات إلى Google Sheets const bookingData = { branch: branchName, date: day, time: time, name: name, phone: phone, notes: notes }; try { await fetch(API_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(bookingData) }); console.log('✅ تم تسجيل الحجز في Google Sheets'); } catch (error) { console.error('❌ خطأ في الإرسال إلى Google Sheets:', error); } // فتح واتساب let message = `مرحباً، أريد حجز موعد في فرع ${branchName}:\n`; message += `📅 اليوم: ${day}\n`; message += `⏰ الوقت: ${time}\n`; message += `👤 الاسم: ${name}\n`; message += `📞 رقم الهاتف: ${phone}\n`; if (notes) { message += `📝 ملاحظات: ${notes}\n`; } message += `\nشكراً!`; const encodedMessage = encodeURIComponent(message); const whatsappUrl = `https://wa.me/${branchPhone}?text=${encodedMessage}`; window.open(whatsappUrl, '_blank'); alert('✅ تم تسجيل حجزك وإرسال رسالة إلى واتساب.'); } // ===== ربط الأحداث عند تحميل الصفحة ===== document.addEventListener('DOMContentLoaded', function() { const branchSelect = document.getElementById('branch'); // عند تغيير الفرع، نجيب المواعيد الخاصة به branchSelect.addEventListener('change', function() { const branchName = this.value; if (branchName) { updateAvailableSlots(branchName); } else { document.getElementById('day').innerHTML = ''; document.getElementById('time').innerHTML = ''; } }); });
