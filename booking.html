// ===== 2. دالة جلب مواعيد العمل ===== function getSchedule() { try { // فتح شيت إدارة الموقع const ss = SpreadsheetApp.openById(SITE_SHEET_ID); const sheet = ss.getSheetByName('إدارة_الموقع'); if (!sheet) { return { error: 'تاب "إدارة_الموقع" غير موجود في شيت الموقع' }; } const rows = sheet.getDataRange().getValues(); const schedule = {}; for (let i = 1; i < rows.length; i++) { const key = rows[i][0]; // العمود A const value = rows[i][1]; // العمود B if (key) { schedule[key] = value; } } // للتأكد من البيانات Logger.log('البيانات المسترجعة: ' + JSON.stringify(schedule)); return schedule; } catch (e) { return { error: e.toString() }; } } // ===== 3. دالة توليد رقم حجز ===== function generateBookingNumber() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const colIndex = headers.indexOf('الحجز') + 1; if (!colIndex) return '001'; const lastRow = sheet.getLastRow(); if (lastRow <= 1) return '001'; const lastNum = sheet.getRange(lastRow, colIndex).getValue(); if (!lastNum) return '001'; const num = parseInt(lastNum) + 1; return String(num).padStart(3, '0'); } // ===== 4. دالة doPost (تسجيل الحجز) ===== function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const colIndex = (name) => headers.indexOf(name) + 1; const colSerial = colIndex('رقم'); const colBranch = colIndex('الفرق'); const colDate = colIndex('التاريخ'); const colTime = colIndex('الوقت'); const colName = colIndex('الإسم'); const colPhone = colIndex('رقم الهاتف'); const colNotes = colIndex('الملاحظات'); const colBookTime = colIndex('وقت الحجز'); const colBookNum = colIndex('الحجز'); const colStatus = colIndex('حالة الحجز'); if (!colSerial || !colBranch || !colDate || !colTime || !colName || !colPhone || !colNotes || !colBookTime || !colBookNum || !colStatus) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'بعض الأعمدة غير موجودة. تأكد من العناوين: رقم, الفرق, التاريخ, الوقت, الإسم, رقم الهاتف, الملاحظات, وقت الحجز, الحجز, حالة الحجز' })); } const now = new Date(); const bookingTime = Utilities.formatDate(now, 'Africa/Cairo', 'yyyy-MM-dd HH:mm:ss'); const lastRow = sheet.getLastRow(); const bookingNumber = generateBookingNumber(); const newRow = []; newRow[colSerial - 1] = lastRow; newRow[colBranch - 1] = data.branch || ''; newRow[colDate - 1] = data.date || ''; newRow[colTime - 1] = data.time || ''; newRow[colName - 1] = data.name || ''; newRow[colPhone - 1] = data.phone || ''; newRow[colNotes - 1] = data.notes || ''; newRow[colBookTime - 1] = bookingTime; newRow[colBookNum - 1] = bookingNumber; newRow[colStatus - 1] = 'انتظار'; sheet.appendRow(newRow); return ContentService.createTextOutput(JSON.stringify({ success: true, bookingNumber: bookingNumber })); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })); } } // ===== 5. دالة doGet (لجلب المواعيد) ===== function doGet(e) { const action = e && e.parameter ? e.parameter.action : null; if (action === 'getSchedule') { const schedule = getSchedule(); return ContentService.createTextOutput(JSON.stringify(schedule)) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'OK', message: 'API شغال' })).setMimeType(ContentService.MimeType.JSON);
