// يمسح أي حاجة مش رقم (مسافات، +، إلخ) if (p.startsWith('0')) { p = '20' + p.substring(1); // يشيل الصفر ويحط كود مصر } else if (!p.startsWith('20')) { p = '20' + p; // لو مكتوب من غير صفر ومن غير كود دولة أصلاً } return p; } function getBookingsSheet() { const ss = SpreadsheetApp.openById(SITE_SHEET_ID); const sheet = ss.getSheetByName(BOOKINGS_SHEET_NAME); if (!sheet) throw new Error('تاب "' + BOOKINGS_SHEET_NAME + '" غير موجود'); i < data.length; i++) { const day = data[i][0]; const octTime = data[i][colOct]; const faiTime = data[i][colFai]; if (day) { if (octTime && octTime !== 'مغلق' && octTime !== '') { scheduleList.push({ branch: "أكتوبر", day: day, time: octTime, status: "مفتوح" }); } if (faiTime && faiTime !== 'مغلق' && faiTime !== '') { scheduleList.push({ branch: "فيصل", day: day, time: faiTime, status: "مفتوح" }); } } } return { status: "OK", data: scheduleList }; } catch (e) { return { error: e.toString() }; } } function generateBookingNumber(sheet, targetDate) { const lastRow = sheet.getLastRow(); if (lastRow <= 1) return '001'; const allData = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); let maxNum = 0; for (let i = 0; i < allData.length; i++) { const rowDate = allData[i][2]; // عمود C (التاريخ) const rowNum = allData[i][8]; // عمود I (رقم الحجز) if (rowDate && rowDate.toString().includes(targetDate)) { const parsedNum = parseInt(rowNum); if (!isNaN(parsedNum) && parsedNum > maxNum) { maxNum = parsedNum; } } } const nextNum = maxNum + 1; return String(nextNum).padStart(3, '0'); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = getBookingsSheet(); // <-- بقى بيفتح تاب "الحجوزات" بالاسم دايمًا، مش أي تاب نشط const now = new Date(); const bookingTime = Utilities.formatDate(now, 'Africa/Cairo', 'yyyy-MM-dd HH:mm:ss'); const lastRow = sheet.getLastRow(); const rawDate = data.date || ''; const matchDate = rawDate.match(/\d{4}-\d{2}-\d{2}/); const targetDateStr = matchDate ? matchDate[0] : rawDate; const bookingNumber = generateBookingNumber(sheet, targetDateStr); // ترتيب الأعمدة: A=رقم, B=الفرع, C=التاريخ, D=الوقت, E=الإسم, F=رقم الهاتف, G=الملاحظات, H=وقت الحجز, I=رقم الحجز, J=حالة الحجز, K=تاريخ التأكيد const newRow = []; newRow[0] = lastRow; // A: رقم newRow[1] = data.branch || ''; // G: الملاحظات newRow[7] = bookingTime; // H: وقت الحجز newRow[8] = bookingNumber; // I: رقم الحجز newRow[9] = 'انتظار'; // J: حالة الحجز // K (تاريخ التأكيد) بتفضل فاضية لحد ما يتم التأكيد const newRowIndex = lastRow + 1; // نحدد عمود F (رقم الهاتف) كنص Text الأول عشان الصفر اللي في الأول ميتمسحش sheet.getRange(newRowIndex, 6).setNumberFormat('@'); sheet.getRange(newRowIndex, 1, 1, newRow.length).setValues([newRow]); return ContentService.createTextOutput(JSON.stringify({ success: true, bookingNumber: bookingNumber })); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })); } } function doGet(e) { const action = e && e.parameter ? e.parameter.action : null;
